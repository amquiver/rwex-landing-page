   150→function generateSlug(text: string): string {
   151→  return text
   152→    .toLowerCase()
   153→    .replace(/[^a-z0-9]+/g, '-')
   154→    .replace(/(^-|-$)/g, '');
   155→}

   157→function formatDate(dateStr: string): string {
   158→  const d = new Date(dateStr);
   159→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
   160→}
   161→
   162→function truncate(str: string, len: number): string {
   163→  if (!str) return '';
   164→  return str.length > len ? str.substring(0, len) + '...' : str;
   165→}

   169→async function apiFetch<T>(url: string, options?: RequestInit): Promise<T> {
   170→  const res = await fetch(url, options);
   171→  if (!res.ok) {
   172→    const data = await res.json().catch(() => ({ error: 'Request failed' }));
   173→    throw new Error(data.error || `HTTP ${res.status}`);
   174→  }
   175→  const json = await res.json();
   176→  // Unwrap pagination so admin code can access data.total directly
   177→  if (json.pagination) {
   178→    return { ...json, ...json.pagination } as T;
   179→  }
   180→  return json as T;
   181→}

   183→async function uploadFile(file: File): Promise<string> {
   184→  const formData = new FormData();
   185→  formData.append('file', file);
   186→  const res = await fetch('/api/admin/upload', { method: 'POST', body: formData });
   187→  if (!res.ok) throw new Error('Upload failed');
   188→  const data = await res.json();
   189→  return data.url;
   190→}
