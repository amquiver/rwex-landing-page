   149→function generateSlug(text: string): string {
   150→  return text
   151→    .toLowerCase()
   152→    .replace(/[^a-z0-9]+/g, '-')
   153→    .replace(/(^-|-$)/g, '');
   154→}

   156→function formatDate(dateStr: string): string {
   157→  const d = new Date(dateStr);
   158→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
   159→}
   160→
   161→function truncate(str: string, len: number): string {
   162→  if (!str) return '';
   163→  return str.length > len ? str.substring(0, len) + '...' : str;
   164→}

   168→async function apiFetch<T>(url: string, options?: RequestInit): Promise<T> {
   169→  const res = await fetch(url, options);
   170→  if (!res.ok) {
   171→    const data = await res.json().catch(() => ({ error: 'Request failed' }));
   172→    throw new Error(data.error || `HTTP ${res.status}`);
   173→  }
   174→  const json = await res.json();
   175→  // Unwrap pagination so admin code can access data.total directly
   176→  if (json.pagination) {
   177→    return { ...json, ...json.pagination } as T;
   178→  }
   179→  return json as T;
   180→}

   182→async function uploadFile(file: File): Promise<string> {
   183→  const formData = new FormData();
   184→  formData.append('file', file);
   185→  const res = await fetch('/api/admin/upload', { method: 'POST', body: formData });
   186→  if (!res.ok) throw new Error('Upload failed');
   187→  const data = await res.json();
   188→  return data.url;
   189→}
